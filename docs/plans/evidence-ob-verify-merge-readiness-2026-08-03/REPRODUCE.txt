# Marker-file rc capture (evidence-13 lesson: do NOT echo $? across a reap boundary;
# capture rc inside the same shell that runs run_all, persist to a file).
# If host UDP 2623 / TCP 2222 / TCP 2445 are free, omit the OB_QEMU_*_PORT overrides.
rm -f /tmp/ob_verify_integ_marker.txt
OB_QEMU_SSH_PORT=32223 OB_QEMU_REDFISH_PORT=32443 OB_QEMU_IPMI_PORT=32623 \
  bash -c 'bash tests/run_all.sh --full --integration
           rc=$?
           {
             echo "INTEGRATION_RC=$rc"
             echo "PIDS_AFTER=$(ls workspace/qemu-bin/.pids/ 2>/dev/null | wc -l)"
           } > /tmp/ob_verify_integ_marker.txt'
cat /tmp/ob_verify_integ_marker.txt   # authoritative rc source (NOT echo $?)
ls workspace/qemu-bin/.pids/          # must be empty
# artifact: 02-runall-full-integration.log (+ marker)
